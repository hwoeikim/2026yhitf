  /* International numbers: keep digits, spaces, hyphens, parentheses and a leading + */
  function formatPhone(value){
    return String(value)
      .replace(/[^\d+\-\s()]/g,'')
      .replace(/(?!^)\+/g,'')
      .slice(0,20);
  }


  function isValidPhone(value){
    const digits = String(value).replace(/\D/g,'');
    return digits.length >= 7 && digits.length <= 15;
  }




  root.addEventListener('input', function(e){


    /* Mobile Number 자동 하이픈 */
    if(e.target.matches('input[type="tel"]')){
      e.target.value = formatPhone(e.target.value);
    }




    /* Date of Birth 연도 4자리 초과 방지 */
    if(e.target.matches('input[name="birth"]')){
      const input = e.target;


      if(input.value){
        const parts = input.value.split('-');


        if(parts[0] && parts[0].length > 4){
          parts[0] = parts[0].slice(0,4);
          input.value = parts.join('-');
        }
      }
    }


  });




  root.addEventListener('change', function(e){


    if(!e.target.matches('input[name="birth"]')) return;


    const input = e.target;
    if(!input.value) return;


    const parts = input.value.split('-');
    const year = parts[0] || '';
    const selected = new Date(input.value + 'T00:00:00');
    const today = new Date();
    today.setHours(0,0,0,0);


    if(year.length !== 4 || Number(year) < 1900){
      input.value = '';
      alert('Please enter the year of birth using exactly 4 digits.');
      input.focus();
      return;
    }


    if(selected > today){
      input.value = '';
      alert('Date of birth cannot be a future date.');
      input.focus();
    }


  });

  /* Back/forward or reload loses in-memory state, so keep the submission state
     (request ID + issued Registration Number) in sessionStorage. The same form content
     then reuses the same request ID and never creates a duplicate row. */
  const SUBMISSION_STATE_KEY = 'YHITF_EN_SUBMISSION_STATE';

  function loadSubmissionState(){
    try {
      const raw = sessionStorage.getItem(SUBMISSION_STATE_KEY);
      return raw ? JSON.parse(raw) : null;
    } catch(e){ return null; }
  }

  function saveSubmissionState(state){
    try { sessionStorage.setItem(SUBMISSION_STATE_KEY, JSON.stringify(state)); } catch(e){}
  }

  function createRequestId(){
    if(window.crypto && typeof window.crypto.randomUUID === 'function'){
      return window.crypto.randomUUID();
    }
    return 'yhitf-' + Date.now() + '-' + Math.random().toString(36).slice(2) + '-' + Math.random().toString(36).slice(2);
  }




  // 네트워크 응답이 유실돼도 동일한 requestId로만 한 번 재확인합니다.
  // 서버가 이미 저장했다면 중복 생성 없이 기존 Registration Number를 돌려줍니다.
  async function postEntryWithSafeRetry(payload){
    let lastError;
    for(let attempt=0; attempt<2; attempt++){
      if(attempt > 0){
        if(saveLoadingDesc){
          saveLoadingDesc.innerHTML = 'Checking the registration result again.<br>A duplicate registration will not be created.';
        }
        await new Promise(function(resolve){ setTimeout(resolve, 800); });
      }
      try {
        const response = await fetch(DB_API_URL, {
          method:'POST',
          headers:{'Content-Type':'text/plain;charset=utf-8'},
          body:JSON.stringify(payload),
          redirect:'follow'
        });
        if(!response.ok){
          const error = new Error('Server response error (' + response.status + ')');
          // 명시적 클라이언트 오류는 재전송하지 않습니다.
          if(response.status >= 400 && response.status < 500) throw Object.assign(error,{noRetry:true});
          throw error;
        }
        // 응답 본문이 유실/손상된 경우에도 같은 요청 ID로만 재확인합니다.
        return await response.json();
      } catch(error){
        lastError = error;
        if(error.noRetry) break;
      }
    }
    throw Object.assign(new Error('We could not confirm the registration result from the server. Your entered information has been preserved. Please click the registration button again to check the result. (' + (lastError && lastError.message || 'Network error') + ')'), {yhNetwork:true});
  }

    // A genuinely changed form is a new submission. Unchanged retries retain the same request ID.

      equipmentAgreed: equipmentAgreed
    };

  function renderEntryPaymentSummary(){
    const breakdown=document.getElementById('entry-summary-breakdown');
    const totalEl=document.getElementById('entry-summary-total');
    const submitBtn=document.getElementById('entry-submit');
    if(!breakdown || !totalEl) return;


    const cards=participantList.querySelectorAll('.participant-card');
    const counts={};
    let total=0;
    let unselected=0;


    cards.forEach(function(card){
      const select=card.querySelector('select[name="participant_course"]');
      if(!select || !select.value){ unselected++; return; }
      const parts=String(select.value).split('||');
      const course=parts[0] || '';
      const date=parts[1] || '';
      const price=getCoursePrice(course);
      const key=course+'||'+date;
      if(!counts[key]) counts[key]={course:course,date:date,count:0,price:price};
      counts[key].count++;
      total+=price;
    });


    /* Race-not-selected indicator: display only, unrelated to amount/saving logic */
    const unselectedEl=document.getElementById('summary-unselected');
    if(unselectedEl){
      unselectedEl.hidden=unselected===0;
      unselectedEl.textContent=unselected>0 ? 'Race not selected: '+unselected : '';
    }

    const items=Object.keys(counts);
    if(!items.length){
      breakdown.innerHTML='<div class="summary-breakdown-row"><span>The estimated payment amount will be shown after selecting a race and date.</span></div>';
    }else{
      breakdown.innerHTML=items.map(function(key){
        const item=counts[key];
        const subtotal=item.price*item.count;
        return '<div class="summary-breakdown-row">'+
          '<span><strong>'+escapeHTML(getImwebOptionLabel(item.course,item.date))+'</strong> × '+item.count+'</span>'+
          '<span>$'+subtotal.toFixed(2)+'</span>'+
        '</div>';
      }).join('');
    }


    totalEl.textContent='$'+total.toFixed(2);
    if(submitBtn){
      submitBtn.textContent=total>0
        ? 'Registration Fee $'+total.toFixed(2)+' USD · Proceed to Payment →'

  function updateTrailAgreementState(){

  function renderPaymentSummary(participants){
    const counts = {};
    let total = 0;


    participants.forEach(function(p){
      const course = String(p.course || '').toUpperCase();
      const price = getCoursePrice(course);
      if(!counts[course]) counts[course] = { count:0, price:price };
      counts[course].count += 1;
      total += price;
    });


    const order = ['5KM RUNNING','10KM RUNNING','15KM TRAIL RUN','28KM TRAIL RUN'];
    const breakdown = document.getElementById('payment-breakdown');
    breakdown.innerHTML = order.filter(function(course){ return counts[course]; }).map(function(course){
      const item = counts[course];
      const subtotal = item.price * item.count;
      return '<div class="payment-breakdown-row">' +
        '<span><strong>' + escapeHTML(getImwebCourseLabel(course)) + '</strong> × ' + item.count + '</span>' +
        '<span>$' + subtotal.toFixed(2) + '</span>' +
      '</div>';
    }).join('');


    document.getElementById('payment-summary-total').textContent = '$'+total.toFixed(2);
  }



  /* =====================================================
     YHITF_ADDRESS (2026-10-08 추가)
     - 주소는 참가신청 서버(앱스크립트)로 보내지 않고, 신청번호가 나온 뒤
       별도 구글폼(주소 응답 탭)으로만 보냅니다.
  ===================================================== */
  const YH_ADDR_FORM_ACTION = "https://docs.google.com/forms/d/e/1FAIpQLSdN7kSxMzEjo4iNCgk2WE85A2d8GH6JTNMVafqG73YPN1GRRw/formResponse";
  const YH_ADDR_ENTRIES = {"appNo":"entry.601683805","name":"entry.344428296","phone":"entry.1313686720","zip":"entry.1661682920","base":"entry.1389995980","detail":"entry.817788371","country":"entry.1043376433","consent":"entry.1181761513"};
  const YH_ADDR_CONSENT = "동의합니다 / I agree";
  let yhAddrInflight = '';
  function yhAddrValue(id){ const el = document.getElementById(id); return el ? String(el.value).trim().replace(/\s+/g, ' ') : ''; }
  function yhAddrData(){
    return { zip: yhAddrValue('yh-addr-zip'), base: [yhAddrValue('yh-addr-base'), yhAddrValue('yh-addr-city')].filter(Boolean).join(', '), detail: yhAddrValue('yh-addr-detail'), country: yhAddrValue('yh-addr-country') };
  }
  function yhAddrText(){ const a = yhAddrData(); return (a.zip ? '(' + a.zip + ') ' : '') + [a.base, a.detail, a.country].filter(Boolean).join(' '); }
  function yhSendAddress(applicationNumber, payload, keepalive){
    try {
      if(!applicationNumber || !payload) return;
      const a = yhAddrData();
      if(!a.zip || !a.base) return;
      const first = (payload.participants || [])[0] || {};
      const isGroup = payload.entryType === 'group';
      const name = isGroup ? ((payload.representative || {}).name || '') : (first.name || [first.firstName, first.lastName].filter(Boolean).join(' '));
      const phone = isGroup ? ((payload.representative || {}).phone || '') : (first.phone || '');
      const key = 'yh_addr_sent_' + applicationNumber;
      const sig = JSON.stringify([a, name, phone]);
      let sent = '';
      try { sent = localStorage.getItem(key) || ''; } catch(e) {}
      if(sent === sig || yhAddrInflight === sig) return;
      const body = new URLSearchParams();
      body.append(YH_ADDR_ENTRIES.appNo, applicationNumber);
      body.append(YH_ADDR_ENTRIES.name, name);
      body.append(YH_ADDR_ENTRIES.phone, phone);
      body.append(YH_ADDR_ENTRIES.zip, a.zip);
      body.append(YH_ADDR_ENTRIES.base, a.base);
      body.append(YH_ADDR_ENTRIES.detail, a.detail);
      body.append(YH_ADDR_ENTRIES.country, a.country);
      body.append(YH_ADDR_ENTRIES.consent, YH_ADDR_CONSENT);
      yhAddrInflight = sig;
      fetch(YH_ADDR_FORM_ACTION, { method: 'POST', mode: 'no-cors', body: body, keepalive: !!keepalive })
        .then(function(){ try { localStorage.setItem(key, sig); } catch(e) {} })
        .catch(function(err){ console.warn('YHITF ADDRESS SEND FAILED', err); })
        .then(function(){ if(yhAddrInflight === sig) yhAddrInflight = ''; });
    } catch(err) { console.warn('YHITF ADDRESS ERROR', err); }
  }

  submitButton.addEventListener(
    'click',
    async function(event){


      event.preventDefault();
      event.stopImmediatePropagation();


      if(submitButton.dataset.saving === 'true'){
        return;
      }




      if(!validateRequiredFields()){
        return;
      }




      const pledgeAgree = document.getElementById('pledge-agree');
      const equipmentAgree = document.getElementById('equipment-agree');
      const privacyAgree = document.getElementById('privacy-agree');
      const hasTrailParticipant = Array.from(participantList.querySelectorAll('select[name="participant_course"]')).some(function(select){
        return /^(15KM|28KM)/.test(String(select.value || '').toUpperCase());
      });

      if(!pledgeAgree.checked){
        alert('Please review and agree to the Participant Agreement.');
        document.querySelector('.pledge-agreement').open = true;
        pledgeAgree.scrollIntoView({behavior:'smooth', block:'center'});
        return;
      }

      if(hasTrailParticipant && !equipmentAgree.checked){
        alert('Please review and agree to the mandatory trail-running equipment rules.');
        document.getElementById('trail-equipment-panel').open = true;
        equipmentAgree.scrollIntoView({behavior:'smooth', block:'center'});
        return;
      }

      if(!privacyAgree.checked){
        alert('Please agree to the collection and use of personal information.');
        return;
      }




      const payload = collectEntryData();
      const currentSignature = pendingPayloadSignature;
      if(completedSubmission && currentSignature === lastCompletedSignature){
        if(savedApplicationNumber && savedPayload){
          showConfirmation(savedApplicationNumber, savedPayload);
        } else {
          alert('This registration has already been saved. Please check your registration number.');
        }
        return;
      }


      const storedState = loadSubmissionState();
      if(storedState && storedState.signature === currentSignature && storedState.applicationNumber){
        completedSubmission = true;
        lastCompletedSignature = currentSignature;
        showConfirmation(storedState.applicationNumber, payload);
        return;
      }


      const originalText = submitButton.textContent;


      submitButton.dataset.saving = 'true';
      submitButton.disabled = true;
      submitButton.textContent = 'Saving registration...';
      submitButton.style.opacity = '.65';
      submitButton.style.cursor = 'wait';
      showSaveLoading();




      try {


        const result = await postEntryWithSafeRetry(payload);




        if(!result.success){
          throw new Error(
            result.message ||
            'Failed to save registration information.'
          );
        }




        // Refuse mismatched server results before showing STEP 02 or enabling payment.
        const expectedPrefix = payload.entryType === 'group' ? 'YHITF-EG-' : 'YHITF-EP-';
        if(typeof result.applicationNumber !== 'string' || !result.applicationNumber.startsWith(expectedPrefix) || Number(result.count) !== payload.participants.length){
          throw new Error('The registration number or participant count does not match the submitted information. Please do not proceed with payment and contact the organizer.');
        }
        completedSubmission = true;
        lastCompletedSignature = currentSignature;
        saveSubmissionState({
          signature: currentSignature,
          requestId: payload.requestId,
          applicationNumber: result.applicationNumber
        });


        /*
          Payment 연결 단계에서 다시 사용할 수 있도록
          Registration Number를 브라우저에 임시 보관
        */
        try {
          sessionStorage.setItem(
            'yhitf_application_number',
            result.applicationNumber
          );
        } catch(e) {}




        hideSaveLoading();
        showConfirmation(result.applicationNumber, payload);




        /*
          다음 단계에서 여기에
          아임웹 Payment페이지 이동 코드를 연결합니다.
        */


      }


      catch(error){
        hideSaveLoading();


        console.error('YHITF ENTRY SAVE ERROR:', error);


        if(error && error.yhNetwork){
          alert("We did not receive a response from the registration server.\n\nYour registration may already have been saved.\nWithout changing anything, please click the [Proceed to Payment] button once more.\nIf it was saved, your existing registration number will be shown and no duplicate will be created.\n\nIf this message keeps appearing:\n· Turn off office/public Wi-Fi and try again with mobile data.\n· If you are in an in-app browser (KakaoTalk, Instagram, etc.), open the site in Chrome or Safari.\n· Temporarily turn off ad blockers or VPN.\n\nIf it still does not work, please leave your name and phone number via 1:1 inquiry and we will check your registration.");
          return;
        }
        alert(
          'Please check your registration result.\n\n' +
          'Your entered information has been preserved. Please review the message and try again.\n' +
          '(' + error.message + ')'
        );


      }


      finally {
        hideSaveLoading();


        submitButton.dataset.saving = 'false';
        submitButton.disabled = false;
        submitButton.textContent = originalText;
        submitButton.style.opacity = '';
        submitButton.style.cursor = '';


      }


    },
    true
  );

  ===================================================== */

  updateTrailAgreementState();




})();
